cmake_minimum_required(VERSION 3.22.1)
project(harness_local LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Native inference requires optimization" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID AND NOT ANDROID_ABI STREQUAL "arm64-v8a" AND NOT ANDROID_ABI STREQUAL "x86_64")
    add_library(harness_local SHARED local_unavailable.cpp)
    target_link_options(harness_local PRIVATE "-Wl,-z,max-page-size=16384")
    return()
endif()
include(FetchContent)
# CPU variants are selected using runtime features, retaining a generic ARM fallback.
if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a")
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
else()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
option(HORIZON_VULKAN "Experimental Vulkan backend; not verified on this phone" OFF)
# Android Vulkan headers and shaders are compiled on the host; no host GPU is used.
if(HORIZON_VULKAN AND ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a")
    FetchContent_Declare(vulkan_headers
        URL https://codeload.github.com/KhronosGroup/Vulkan-Headers/tar.gz/c46850864f4661461b0f6cb9922c058ffea4915e
        URL_HASH SHA256=539ff8eaa4abf22fddbd98225c68a0c5ddb940075a4048c8b7f5d04449070a4e)
    FetchContent_GetProperties(vulkan_headers)
    if(NOT vulkan_headers_POPULATED)
        FetchContent_Populate(vulkan_headers)
    endif()
    FetchContent_Declare(spirv_headers
        URL https://codeload.github.com/KhronosGroup/SPIRV-Headers/tar.gz/86f980c731e62ae4eaf383d320449d71687936bf
        URL_HASH SHA256=d11ef011d0788ed002d7cc96224d298f43b126cabe887cd4d46199cb19ecf386)
    FetchContent_GetProperties(spirv_headers)
    if(NOT spirv_headers_POPULATED)
        FetchContent_Populate(spirv_headers)
    endif()
    set(Vulkan_INCLUDE_DIR "${vulkan_headers_SOURCE_DIR}/include" CACHE PATH "" FORCE)
    set(Vulkan_LIBRARY "${CMAKE_ANDROID_NDK}/toolchains/llvm/prebuilt/${ANDROID_HOST_TAG}/sysroot/usr/lib/aarch64-linux-android/28/libvulkan.so" CACHE FILEPATH "" FORCE)
    if(CMAKE_HOST_WIN32)
        set(Vulkan_GLSLC_EXECUTABLE "${CMAKE_ANDROID_NDK}/shader-tools/windows-x86_64/glslc.exe" CACHE FILEPATH "" FORCE)
    else()
        find_program(Vulkan_GLSLC_EXECUTABLE glslc HINTS "${CMAKE_ANDROID_NDK}/shader-tools/${ANDROID_HOST_TAG}" NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    endif()
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/spirv-config")
    file(WRITE "${CMAKE_BINARY_DIR}/spirv-config/SPIRV-HeadersConfig.cmake"
        "if(NOT TARGET SPIRV-Headers::SPIRV-Headers)\nadd_library(SPIRV-Headers::SPIRV-Headers INTERFACE IMPORTED)\nset_target_properties(SPIRV-Headers::SPIRV-Headers PROPERTIES INTERFACE_INCLUDE_DIRECTORIES \"${spirv_headers_SOURCE_DIR}/include\")\nendif()\n")
    include_directories("${spirv_headers_SOURCE_DIR}/include")
    set(SPIRV-Headers_DIR "${CMAKE_BINARY_DIR}/spirv-config" CACHE PATH "" FORCE)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
else()
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
endif()
FetchContent_Declare(llama
    URL https://codeload.github.com/ifm-ai/llama.cpp/tar.gz/42adf019f76013dac873b5b43950d54d5ab27216
    URL_HASH SHA256=c58cab48ce95510c65ed7f7abe20a3c70c5a0874908dd48268daa552d348aabb
)
FetchContent_MakeAvailable(llama)
add_library(harness_local SHARED local_model.cpp)
target_link_libraries(harness_local PRIVATE llama mtmd)
if(ANDROID)
    target_link_libraries(harness_local PRIVATE android log)
else()
    find_package(JNI REQUIRED)
    target_include_directories(harness_local PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_link_options(harness_local PRIVATE "-Wl,-z,max-page-size=16384")
