cmake_minimum_required(VERSION 3.22.1)
project(horizon_mnn LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized native bridge" FORCE)
add_library(MNN SHARED IMPORTED)
set_target_properties(MNN PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/mnn-sdk/${ANDROID_ABI}/libMNN.so")
add_library(harness_local SHARED mnn_local.cpp)
target_include_directories(harness_local PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mnn-sdk/include")
target_compile_options(harness_local PRIVATE -O3)
target_link_libraries(harness_local PRIVATE MNN android log)
target_link_options(harness_local PRIVATE "-Wl,-z,max-page-size=16384")
